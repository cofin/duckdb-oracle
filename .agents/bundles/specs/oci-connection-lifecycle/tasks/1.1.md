### Step 1: Replace global Windows `setenv`/`unsetenv` shims and add `RedactOracleConnectionString` in `src/include/oracle_utils.hpp`

In `src/include/oracle_utils.hpp:8-20`, remove the global `setenv`/`unsetenv` functions and place `SetOracleEnvVar`, `UnsetOracleEnvVar`, and `RedactOracleConnectionString` inside `namespace duckdb`:

```cpp
namespace duckdb {

inline int SetOracleEnvVar(const char *name, const char *value) {
#ifdef _WIN32
	std::string entry = std::string(name ? name : "") + "=" + std::string(value ? value : "");
	return _putenv(entry.c_str());
#else
	return setenv(name, value ? value : "", 1);
#endif
}

inline int UnsetOracleEnvVar(const char *name) {
#ifdef _WIN32
	return _putenv_s(name, "");
#else
	return unsetenv(name);
#endif
}

inline std::string RedactOracleConnectionString(const std::string &conn_str) {
	auto slash_pos = conn_str.find('/');
	auto at_pos = conn_str.rfind('@');
	if (slash_pos == std::string::npos || at_pos == std::string::npos || slash_pos >= at_pos) {
		return conn_str;
	}
	if (slash_pos == 0 && at_pos == 1) {
		return conn_str;
	}
	return conn_str.substr(0, slash_pos + 1) + "***" + conn_str.substr(at_pos);
}
```

### Step 2: Zero-initialize `errbuf` and check `OCIErrorGet` return code in `OCIErrorCodeMatches` and `CheckOCIError`

In `src/include/oracle_utils.hpp:62-85`, update `OCIErrorCodeMatches` and `CheckOCIError`:

```cpp
inline bool OCIErrorCodeMatches(OCIError *errhp, sb4 expected_code) {
	if (!errhp) {
		return false;
	}
	text errbuf[512] = {0};
	sb4 errcode = 0;
	sword rc = OCIErrorGet((dvoid *)errhp, (ub4)1, (text *)NULL, &errcode, errbuf, (ub4)sizeof(errbuf),
	                       OCI_HTYPE_ERROR);
	return rc == OCI_SUCCESS && errcode == expected_code;
}

inline void CheckOCIError(sword status, OCIError *errhp, const std::string &msg) {
	if (status != OCI_SUCCESS && status != OCI_SUCCESS_WITH_INFO) {
		if (status == OCI_INVALID_HANDLE || !errhp) {
			throw IOException("%s: Invalid OCI handle (status=%d)", msg, static_cast<int>(status));
		}
		text errbuf[512] = {0};
		sb4 errcode = 0;
		sword get_rc = OCIErrorGet((dvoid *)errhp, (ub4)1, (text *)NULL, &errcode, errbuf, (ub4)sizeof(errbuf),
		                           OCI_HTYPE_ERROR);
		if (get_rc != OCI_SUCCESS) {
			throw IOException("%s: OCI call failed with status %d (OCIErrorGet returned %d)", msg,
			                  static_cast<int>(status), static_cast<int>(get_rc));
		}
		throw IOException("%s: %s (Error code: %d)", msg, std::string(reinterpret_cast<char *>(errbuf)), errcode);
	}
}
```
